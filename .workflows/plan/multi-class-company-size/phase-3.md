```python
"""Phase 3 checks for the multi-class company-size refresh. Read-only on every store.

measure STORE OUT.json [--from D] [--to D] [--values]
    per-year coverage of index member-days (share count, close, market value), the census of
    member-days with a close but no share count per symbol, the done-means names' gap days, and
    with --values the spot values on the last session on or before 2014-06-30.
compare BEFORE.json AFTER.json
    the before/after table and PASS/FAIL lines; exit 1 on any FAIL.
rows OLD_SHARES NEW_SHARES CLASS_TABLE
    which symbols' share values changed, which of those have no per-class row and no class-table
    row (unexpected: exit 1), Apple untouched, and new rows that jump more than 3x.
manifests OLD_MANIFEST NEW_MANIFEST
    which store files changed; exit 1 unless only shares.csv (and splits.csv, warned) did.
"""

from __future__ import annotations

import argparse
import csv
import json
import sys
from collections import Counter, defaultdict
from datetime import date
from pathlib import Path

from seer_engine import dates, research

DONE_MEANS = ("META", "FB", "V", "MA", "GOOGL", "GOOG", "BRK.B")
SPOT_DAY = date(2014, 6, 30)
# label -> (symbols summed over those that are members that day, low $B, high $B); None = info only
SPOT = {
    "Apple": (("AAPL",), 540.0, 580.0),
    "Meta": (("META", "FB"), 160.0, 190.0),
    "Visa": (("V",), 120.0, 140.0),
    "Berkshire": (("BRK.B",), 290.0, 335.0),
    "Alphabet (GOOGL+GOOG)": (("GOOGL", "GOOG"), 360.0, 420.0),
    "Mastercard": (("MA",), None, None),
    "Comcast": (("CMCSA",), None, None),
    "UPS": (("UPS",), None, None),
    "Ford": (("F",), None, None),
}
# US filers with more than one class (analysis gap census): left over gap days above this are
# explained one by one in the handover close-out.
WATCH = (
    "ACN", "MKC", "LEN", "CMCSA", "STZ", "EXPE", "CME", "EL", "WBD", "MA", "TAP", "TSN", "UPS",
    "RL", "V", "HRL", "BRK.B", "F", "META", "GHC", "NWSA", "NWS", "FOXA", "FOX", "LBTYA",
    "LBTYK", "UAA", "BF.B", "GOOGL", "GOOG", "LILA", "LILAK",
)
WATCH_MAX_GAP = 30
CLASS_SOURCE = "xbrl-class"
JUMP = 3.0


def measure(args: argparse.Namespace) -> int:
    store = Path(args.store)
    window = research.declared_window(store)
    data = research.load_store(store, window=window)
    m = data.market
    etfs = set(research.RESEARCH_ETFS)
    start = args.start
    end = min(args.end, data.window.end) if args.end else data.window.end
    years: dict[int, list[int]] = defaultdict(lambda: [0, 0, 0, 0])
    gaps: Counter[str] = Counter()
    first_gap: dict[str, str] = {}
    last_gap: dict[str, str] = {}
    close_days: Counter[str] = Counter()
    for d in dates.sessions(start, end):
        row = years[d.year]
        for s in m.membership.members_on(d):
            if s in etfs:
                continue
            h = m.history.get(s)
            has_close = h is not None and h.index_of(d) is not None
            has_shares = m.shares_on(s, d) is not None
            row[0] += 1
            row[1] += has_shares
            row[2] += has_close
            row[3] += m.market_value(s, d) is not None
            if has_close:
                close_days[s] += 1
                if not has_shares:
                    gaps[s] += 1
                    first_gap.setdefault(s, d.isoformat())
                    last_gap[s] = d.isoformat()
    spot: dict[str, dict] = {}
    if args.values:
        day = dates.sessions(date(SPOT_DAY.year, 1, 1), SPOT_DAY)[-1]
        members = m.membership.members_on(day)
        for label, (syms, lo, hi) in SPOT.items():
            parts = {s: m.market_value(s, day) for s in syms if s in members}
            known = bool(parts) and all(v is not None for v in parts.values())
            spot[label] = {
                "day": day.isoformat(),
                "parts_billions": {s: (None if v is None else round(v / 1e9, 2)) for s, v in parts.items()},
                "shares": {s: m.shares_on(s, day) for s in parts},
                "billions": round(sum(parts.values()) / 1e9, 2) if known else None,
                "low": lo,
                "high": hi,
            }
    out = {
        "store": str(store),
        "fingerprint": data.fingerprint,
        "price_fingerprint": data.price_fingerprint,
        "window": [data.window.name, data.window.start.isoformat(), data.window.end.isoformat()],
        "from": start.isoformat(),
        "to": end.isoformat(),
        "years": {str(y): v for y, v in sorted(years.items())},
        "gap_total": sum(gaps.values()),
        "gaps": dict(gaps.most_common()),
        "first_gap": first_gap,
        "last_gap": last_gap,
        "done_means": {
            s: {"close_days": close_days[s], "gap_days": gaps[s], "first_gap": first_gap.get(s),
                "last_gap": last_gap.get(s)}
            for s in DONE_MEANS
        },
        "spot": spot,
    }
    Path(args.out).write_text(json.dumps(out, indent=1, sort_keys=True), encoding="utf-8")
    print(f"{store}: fingerprint {data.fingerprint[:8]}, price fingerprint {data.price_fingerprint[:8]}")
    print(f"{'year':>4} {'member-days':>11} {'shares':>8} {'close':>8} {'mkt value':>9}")
    for y, (n, sh, cl, mv) in sorted(years.items()):
        print(f"{y:>4} {n:>11} {sh / n:>8.1%} {cl / n:>8.1%} {mv / n:>9.1%}")
    print(f"close but no share count: {len(gaps)} symbols, {sum(gaps.values())} member-days")
    for s, n in gaps.most_common(40):
        print(f"  {s:<7} {n:>5}  {first_gap[s]} .. {last_gap[s]}")
    for label, v in spot.items():
        print(f"  {label:<24} {v['billions']}B  {v['parts_billions']}")
    return 0


def _pct(n: int, d: int) -> str:
    return f"{n / d:.1%}" if d else "-"


def compare(args: argparse.Namespace) -> int:
    b = json.loads(Path(args.before).read_text(encoding="utf-8"))
    a = json.loads(Path(args.after).read_text(encoding="utf-8"))
    fails: list[str] = []
    warns: list[str] = []
    if b["price_fingerprint"] != a["price_fingerprint"]:
        fails.append(f"price fingerprint moved {b['price_fingerprint']} -> {a['price_fingerprint']}")
    print(f"fingerprint {b['fingerprint'][:8]} -> {a['fingerprint'][:8]}, price fingerprint "
          f"{b['price_fingerprint'][:8]} -> {a['price_fingerprint'][:8]}")
    print(f"{'year':>4} {'shares before':>13} {'after':>7} {'mkt value before':>16} {'after':>7}")
    for y in sorted(a["years"]):
        nb, shb, clb, mvb = b["years"][y]
        na, sha, cla, mva = a["years"][y]
        if (nb, clb) != (na, cla):
            fails.append(f"{y}: member-days or closes moved ({nb}, {clb}) -> ({na}, {cla})")
        if sha < shb:
            fails.append(f"{y}: share-count coverage fell {_pct(shb, nb)} -> {_pct(sha, na)}")
        if mva < mvb:
            fails.append(f"{y}: market-value coverage fell {_pct(mvb, nb)} -> {_pct(mva, na)}")
        print(f"{y:>4} {_pct(shb, nb):>13} {_pct(sha, na):>7} {_pct(mvb, nb):>16} {_pct(mva, na):>7}")
    print(f"close but no share count: {b['gap_total']} -> {a['gap_total']} member-days, "
          f"{len(b['gaps'])} -> {len(a['gaps'])} symbols")
    if b["gap_total"] and a["gap_total"] >= b["gap_total"]:
        fails.append("the close-but-no-count census did not shrink")
    for s, v in a["done_means"].items():
        if v["close_days"] and v["gap_days"]:
            fails.append(f"{s}: {v['gap_days']} member-days with a close and no count "
                         f"({v['first_gap']} .. {v['last_gap']})")
        elif v["close_days"]:
            print(f"  continuous: {s} ({v['close_days']} member-days with a close)")
    for s in WATCH:
        n = a["gaps"].get(s, 0)
        if n > WATCH_MAX_GAP:
            warns.append(f"{s}: {n} gap days left ({a['first_gap'][s]} .. {a['last_gap'][s]}), "
                         f"was {b['gaps'].get(s, 0)} -- explain in the handover close-out")
    for label, v in a["spot"].items():
        before = b["spot"].get(label, {}).get("billions")
        got = v["billions"]
        band = "" if v["low"] is None else f" band {v['low']:.0f}-{v['high']:.0f}"
        print(f"  spot {v['day']} {label:<24} {before} -> {got} $B{band}  {v['parts_billions']}")
        if v["low"] is not None and (got is None or not v["low"] <= got <= v["high"]):
            fails.append(f"spot {label}: {got} $B outside {v['low']:.0f}-{v['high']:.0f}")
        if label == "Apple" and before is not None and got != before:
            fails.append(f"spot Apple moved {before} -> {got} (single-class filer must not move)")
    for w in warns:
        print("WARN " + w)
    for f in fails:
        print("FAIL " + f)
    print("PASS" if not fails else f"{len(fails)} FAIL")
    return 1 if fails else 0


def _read_rows(path: Path) -> dict[str, list[dict[str, str]]]:
    out: dict[str, list[dict[str, str]]] = defaultdict(list)
    with open(path, newline="", encoding="utf-8") as fh:
        for r in csv.DictReader(fh):
            out[r["symbol"]].append(r)
    return out


def _values(rs: list[dict[str, str]]) -> list[tuple[str, str, str, str]]:
    return [(r["from"], r["shares"], r["counted"], r["filed"]) for r in rs]


def rows(args: argparse.Namespace) -> int:
    old, new = _read_rows(Path(args.old)), _read_rows(Path(args.new))
    table: set[str] = set()
    tpath = Path(args.class_table)
    if tpath.is_file():
        with open(tpath, newline="", encoding="utf-8") as fh:
            table = {r["symbol"] for r in csv.DictReader(line for line in fh if not line.startswith("#"))}
    syms = sorted(set(old) | set(new))
    value_changed = [s for s in syms if _values(old.get(s, [])) != _values(new.get(s, []))]
    source_only = [s for s in syms if s not in value_changed and old.get(s) != new.get(s)]
    per_class = {s for s, rs in new.items() if any(r["source"] == CLASS_SOURCE for r in rs)}
    unexpected = [s for s in value_changed if s not in per_class and s not in table]
    print(f"{len(syms)} symbols; values changed for {len(value_changed)}, source/accession only for "
          f"{len(source_only)}; {len(per_class)} symbols carry a per-class row")
    print("values changed: " + " ".join(value_changed))
    fails = []
    if "AAPL" in value_changed:
        fails.append("AAPL's share values changed")
    if unexpected:
        fails.append("values changed with no per-class row and no class-table row: " + " ".join(unexpected))
    jumps = []
    for s in value_changed:
        old_keys = set(_values(old.get(s, [])))
        known = [r for r in new[s] if r["shares"]]
        for prev, cur in zip(known, known[1:]):
            if (cur["from"], cur["shares"], cur["counted"], cur["filed"]) in old_keys:
                continue
            ratio = float(cur["shares"]) / float(prev["shares"])
            if ratio > JUMP or ratio < 1 / JUMP:
                jumps.append(f"  {s:<7} {prev['from']} {float(prev['shares']):>16,.0f} -> {cur['from']} "
                             f"{float(cur['shares']):>16,.0f} (x{ratio:.2f}, {cur['source']} {cur['accession']})")
    if jumps:
        print(f"{len(jumps)} new rows jump more than {JUMP:g}x from the row before "
              "(check each against splits.csv and the filing):")
        print("\n".join(jumps))
    for f in fails:
        print("FAIL " + f)
    print("PASS" if not fails else f"{len(fails)} FAIL")
    return 1 if fails else 0


def manifests(args: argparse.Namespace) -> int:
    o = json.loads(Path(args.old).read_text(encoding="utf-8"))
    n = json.loads(Path(args.new).read_text(encoding="utf-8"))
    of, nf = o["files"], n["files"]
    changed = sorted(k for k in set(of) | set(nf) if of.get(k) != nf.get(k))
    print(f"fingerprint {o['fingerprint'][:8]} -> {n['fingerprint'][:8]}; files changed: {changed}")
    print(f"price fingerprint {research.price_fingerprint_of(of)[:8]} -> {research.price_fingerprint_of(nf)[:8]}")
    bad = sorted(set(changed) - {research.SHARES_FILE, research.SPLITS_FILE})
    if research.SPLITS_FILE in changed:
        print("WARN splits.csv changed: the split cache moved since the last write; diff it")
```

```python
Why this exists. Company size (market value = shares outstanding x the *unadjusted* close) is
what a market-shaped, size-weighted book ranks on, and the lab could not compute it before
2009: XBRL's ``dei:EntityCommonStockSharesOutstanding`` starts with the large accelerated
filers' fiscal 2009 reports, and EODHD's fundamentals endpoint is not on the owner's plan
(it answers 403). Every 10-K and 10-Q filed since EDGAR began carries the number on its cover
page instead -- "As of March 2, 1997, 1,640,183,273 shares of common stock were outstanding" --
so it is rebuilt here from the free filings, the way ``sectors`` rebuilt industry labels.

Why 2009+ is read from the filings too. SEC's ``companyfacts`` API, which fills a store's
``fundamentals.csv``, drops every dimensional fact, and a company with more than one share class
tags its cover count once per class on the class axis with no total. Meta, Visa, Mastercard,
Comcast, UPS, Ford and Alphabet from 2014 therefore had no 2009+ size at all, and Berkshire
counted its class A only, so every market-shaped book started 2009-2015 behind SPY before it
picked anything. Each filing's own XBRL instance carries the per-class facts; they are read
point in time from the filing date, summed or attributed per listed class by a small class
table, and vendored like the cover-page csv.

* ``edgar_index`` -- EDGAR's quarterly ``master.idx`` as filing references, and which filings
  are wanted for which filer over which window (pure).
* ``fetch`` -- the downloader: index files, the head of each filing's primary document and the
  dei share facts of each XBRL instance into a gitignored on-disk cache, rate limited under
  SEC's ceiling, resumable (network edge).
* ``cover`` -- the cover-page parser: ``(as_of, shares, class)`` out of free text (pure).
* ``instance`` -- the dei share facts and their class-axis contexts out of an XBRL instance:
  one reading per filing, the tagged total if any and the count per class (pure).
* ``classes`` -- which filings need an instance read, ``shares download-classes`` /
  ``build-classes`` and the vendored ``engine/data/shares_xbrl_classes.csv`` (network edge for
  the download, pure for the build and the reader).
* ``class_table`` -- the hand-kept ``engine/data/share_classes.csv``: which listed symbol each
  class of a multi-class filer trades as, at what ratio, keyed on the count's date; classes
  matched by their words, so cover text and XBRL member names share one row (pure).
* ``history`` -- the vendored results (``shares_outstanding_pre2009.csv``,
  ``shares_xbrl_classes.csv``) read through the class table, the point-in-time lookup, the join
  to 2009+ XBRL facts, the per-class attribution and the market-value helper (pure).
* ``checks`` -- the validation that makes the cover numbers usable: XBRL overlap, split
  consistency, implied market value bounds (pure).
* ``splits`` -- the vendor split lists behind the research stores' adjusted closes, cached from
  yfinance (network edge).
```
